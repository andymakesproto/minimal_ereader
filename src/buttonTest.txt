#include <Arduino.h>
#include "InputController.h"

// Instantiate the driver
InputController input;

void setup() {
    // Start the serial monitor at standard ESP32 speed
    Serial.begin(115200);
    delay(2000); // Wait 2 seconds for the USB serial to connect
    Serial.println("--- Booting E-Reader Input Test ---");

    // Initialize the buttons
    input.begin();

    // Attach a temporary callback to print to the screen instead of routing to AppManager
    input.setEventCallback([](int btnId, int eventType) {
        Serial.print("Button Event -> ID: ");
        Serial.print(btnId);
        
        if (eventType == EVENT_CLICK) {
            Serial.println(" | Type: CLICK");
        } else if (eventType == EVENT_LONG_PRESS) {
            Serial.println(" | Type: LONG PRESS");
        }
    });

    Serial.println("Hardware ready. Press some buttons!");
}

void loop() {
    // Keep checking the physical pins
    input.update();
}
